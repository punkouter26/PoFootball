# Version contract (UNITY_RULES §4): the Python trainer and the C# package must
# agree on the communication protocol or the handshake is refused.
#
#   com.unity.ml-agents  4.1.0   ->  comms API 1.5.0
#   mlagents             1.1.0   ->  comms API 1.5.0     <- equal, handshake OK
#
# 4.1.0 is newer than the release_23 tag (4.0.0) but did not move the protocol.
# If you bump either side, re-check Academy.cs `k_ApiVersion` against
# mlagents_envs.environment.UnityEnvironment.API_VERSION before training.

# --- Trainer -----------------------------------------------------------------
mlagents==1.1.0
mlagents-envs==1.1.0

# --- Torch (CUDA 12.1 build) -------------------------------------------------
# Needs the PyTorch index; the plain PyPI wheel is CPU-only.
#
# NOTE (2026-08-20): this machine now has an RTX 5070 Ti Laptop (Blackwell, sm_120).
# cu121 ships no kernels for it. torch.cuda.is_available() still returns True, so the
# trainer picks CUDA and dies on the first step with
#   "CUDA error: no kernel image is available for execution on the device".
# Set CUDA_VISIBLE_DEVICES=-1 before every mlagents-learn run. That is the only thing
# that works: --torch-device=cpu and torch_settings.device: cpu were both tried and both
# still crashed, because mlagents/torch_utils/torch.py calls set_torch_config(device=None)
# at import time, resolves it to "cuda", and its cpu branch never resets the default
# device afterwards. Use -1, not "" — an empty value is ignored on Windows.
#
# To train on the GPU instead, replace this pin with a CUDA 12.8 build (torch>=2.7) —
# mlagents 1.1.0 only requires torch>=1.13.1, so a newer version is allowed.
torch==2.5.1+cu121

# --- Pins that are load-bearing, not incidental ------------------------------
# setuptools 81 removed pkg_resources, which mlagents.torch_utils imports at
# module scope. Anything >=81 breaks `mlagents-learn` on startup.
setuptools<81

# mlagents hard-pins these; listed so a resolver cannot drift them upward.
numpy==1.23.5
protobuf==3.20.3
grpcio==1.48.2
onnx==1.15.0
